#### 2. 校验规则引擎
```java
@Component
public class ValidationRuleEngine {
    
    public ValidationResult validateRow(Map<String, Object> rowData, 
                                      List<ColumnRule> rules) {
        ValidationResult result = new ValidationResult();
        
        for (ColumnRule rule : rules) {
            ValidationError error = validateField(rowData, rule);
            if (error != null) {
                result.addError(error);
                result.addSuggestion(generateSuggestion(error, rule));
            }
        }
        
        return result;
    }
    
    private RepairSuggestion generateSuggestion(ValidationError error, ColumnRule rule) {
        // 根据错误类型生成具体的修复建议
        switch (error.getType()) {
            case REQUIRED_FIELD_MISSING:
                return new RepairSuggestion("请填写必填字段: " + rule.getColumnName());
            case INVALID_FORMAT:
                return new RepairSuggestion("格式错误，期望格式: " + rule.getFormat());
            case VALUE_OUT_OF_RANGE:
                return new RepairSuggestion("值超出范围，有效范围: " + rule.getRange());
            default:
                return new RepairSuggestion("请检查数据格式");
        }
    }
}
```

#### 1. 实时校验组件
```typescript
import React, { useState, useEffect } from 'react';
import { Progress, Alert, Table, Button } from 'antd';
import { useWebSocket } from '@/hooks/useWebSocket';

interface ValidationPanelProps {
  importId: string;
  onValidationComplete: (result: ValidationResult) => void;
}

const ValidationPanel: React.FC<ValidationPanelProps> = ({ importId, onValidationComplete }) => {
  const [progress, setProgress] = useState(0);
  const [errors, setErrors] = useState<ValidationError[]>([]);
  const [suggestions, setSuggestions] = useState<RepairSuggestion[]>([]);
  
  const { connect, disconnect, sendMessage } = useWebSocket({
    url: `/ws/validation/${importId}`,
    onMessage: (data: ValidationFeedback) => {
      setProgress(data.progress);
      setErrors(data.errors);
      setSuggestions(data.suggestions);
      
      if (data.progress === 100) {
        onValidationComplete(data);
      }
    }
  });
  
  useEffect(() => {
    connect();
    return () => disconnect();
  }, []);
  
  const handleAutoFix = (errorId: string) => {
    sendMessage({
      action: 'AUTO_FIX',
      errorId: errorId
    });
  };
  
  return (
    <div className="validation-panel">
      <Progress percent={progress} status={progress === 100 ? 'success' : 'active'} />
      
      {errors.length > 0 && (
        <Alert
          message={`发现 ${errors.length} 个问题`}
          type="warning"
          showIcon
          style={{ margin: '16px 0' }}
        />
      )}
      
      <Table
        dataSource={errors}
        columns={[
          { title: '行号', dataIndex: 'row', key: 'row' },
          { title: '列名', dataIndex: 'column', key: 'column' },
          { title: '错误信息', dataIndex: 'message', key: 'message' },
          { title: '建议', dataIndex: 'suggestion', key: 'suggestion' },
          {
            title: '操作',
            key: 'action',
            render: (_, record) => (
              record.canAutoFix && (
                <Button size="small" onClick={() => handleAutoFix(record.id)}>
                  自动修复
                </Button>
              )
            )
          }
        ]}
        pagination={{ pageSize: 10 }}
      />
    </div>
  );
};

export default ValidationPanel;
```

#### 2. WebSocket Hook
```typescript
import { useRef, useCallback } from 'react';

interface UseWebSocketOptions {
  url: string;
  onMessage: (data: any) => void;
  onError?: (error: Event) => void;
  onClose?: (event: CloseEvent) => void;
}

export const useWebSocket = ({ url, onMessage, onError, onClose }: UseWebSocketOptions) => {
  const wsRef = useRef<WebSocket | null>(null);
  
  const connect = useCallback(() => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      return;
    }
    
    wsRef.current = new WebSocket(`ws://localhost:8080${url}`);
    
    wsRef.current.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data);
        onMessage(data);
      } catch (error) {
        console.error('WebSocket message parse error:', error);
      }
    };
    
    wsRef.current.onerror = onError || ((error) => {
      console.error('WebSocket error:', error);
    });
    
    wsRef.current.onclose = onClose || ((event) => {
      console.log('WebSocket closed:', event);
    });
  }, [url, onMessage, onError, onClose]);
  
  const disconnect = useCallback(() => {
    if (wsRef.current) {
      wsRef.current.close();
      wsRef.current = null;
    }
  }, []);
  
  const sendMessage = useCallback((data: any) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(data));
    }
  }, []);
  
  return { connect, disconnect, sendMessage };
};
```
